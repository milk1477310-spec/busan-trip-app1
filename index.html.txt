                const closeExpenseModal = () => { showExpenseModal.value = false; };
                const saveExpense = () => {
                    if (!newExpense.value.name || !newExpense.value.amount) return alert('請填寫名稱與金額');
                    expenses.value.unshift({ ...newExpense.value });
                    closeExpenseModal();
                };
                const removeExpense = (index) => { expenses.value.splice(index, 1); };

                // 樣式輔助
                const getCategoryBadgeClass = (cat) => {
                    switch(cat) {
                        case '景點': return 'bg-emerald-100 text-emerald-700';
                        case '美食': return 'bg-amber-100 text-amber-700';
                        case '住宿': return 'bg-purple-100 text-purple-700';
                        case '航班': return 'bg-sky-100 text-sky-700';
                        default: return 'bg-slate-100 text-slate-700';
                    }
                };

                const getTransitIcon = (type) => {
                    switch(type) {
                        case '步行': return 'fa-solid fa-person-walking';
                        case '開車': return 'fa-solid fa-car';
                        case '大眾運輸': return 'fa-solid fa-bus';
                        default: return 'fa-solid fa-arrow-right';
                    }
                };

                const getExpenseIcon = (cat) => {
                    switch(cat) {
                        case '交通': return 'fa-solid fa-train-subway';
                        case '飲食': return 'fa-solid fa-utensils';
                        case '住宿': return 'fa-solid fa-hotel';
                        case '門票': return 'fa-solid fa-ticket';
                        case '購物': return 'fa-solid fa-bag-shopping';
                        default: return 'fa-solid fa-wallet';
                    }
                };

                return {
                    currentTab, selectedDay, days, currentWeather, categoryFilter,
                    outboundFlight, returnFlight, hotel, shoppingList, newShoppingItem,
                    expenses, krwInput, calculatedTwd, completedShoppingCount, totalExpenseKrw,
                    filteredSchedules, showModal, isEditing, currentFormItem,
                    showDeleteConfirm, showExpenseModal, newExpense,
                    openAddModal, openEditModal, closeModal, confirmDeleteSchedule, deleteScheduleConfirmed,
                    addShoppingItem, removeShoppingItem, openAddExpenseModal, closeExpenseModal, saveExpense, removeExpense,
                    getCategoryBadgeClass, getTransitIcon, getExpenseIcon
                };
            }
        }).mount('#app');
    </script>
